$auditContent = @'
Write-Host "`n========================================================" -ForegroundColor Cyan
Write-Host "   MARIAN MADRID 5000 - AUDITORIA ESTATICA (src/)       " -ForegroundColor Cyan
Write-Host "========================================================`n" -ForegroundColor Cyan

$jsAudit = path => {
    const fs = require('fs');
    const pathMod = require('path');
    const { execSync } = require('child_process');
    let root = pathMod.join(process.cwd(), 'src');
    if (!fs.existsSync(root)) root = process.cwd();
    
    function getFiles(dir, acc = []) {
        fs.readdirSync(dir).forEach(f => {
            let fp = pathMod.join(dir, f);
            if (fs.statSync(fp).isDirectory() && f !== 'node_modules' && f !== '.git') getFiles(fp, acc);
            else if (f.endsWith('.js')) acc.push(fp);
        });
        return acc;
    }
    const files = getFiles(root);
    console.log('-> Analizando ' + files.length + ' modulos en src/...\n');
    let errors = 0;
    files.forEach(f => {
        try { execSync('node --input-type=module --check', { input: fs.readFileSync(f), stdio: 'pipe' }); }
        catch(e) { console.log('ERROR: ' + pathMod.relative(root, f)); errors++; }
    });
    console.log(errors === 0 ? '\x1b[32m[PASS 100%] 0 Errores de sintaxis.\x1b[0m\n' : '\x1b[31mErrores: ' + errors + '\x1b[0m\n');
};

node -e "($jsAudit)()"
'@
[System.IO.File]::WriteAllText("$PWD\run-velo-audit.ps1", $auditContent, [System.Text.Encoding]::UTF8)
Write-Host "-> run-velo-audit.ps1 guardado en disco." -ForegroundColor Green
